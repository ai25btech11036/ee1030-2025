\begin{frame}{question}
\textbf{Q.}\;
\noindent \textbf{Question 5.7.9:}  
If 
\[
A = \begin{bmatrix}
-3 & 6 \\
-2 & 4
\end{bmatrix},
\]
show that \(A^3 = A\).
\renewcommand{\theequation}{\arabic{equation}}
\setcounter{equation}{0}

\end{frame}
\begin{frame}{solution}
\noindent\textbf{Given}: \(\vec{A}=\myvec{-3&6\\-2&4}\). \quad 
\noindent The characteristic equation is
\begin{align}
f(\lambda)
= \abs{\vec{A}-\lambda\vec{I}}
= \abs{\myvec{-3-\lambda&6\\-2&4-\lambda}}
= (-3-\lambda)(4-\lambda)-(-12)
= \lambda^2-\lambda
= \lambda(\lambda-1)=0.
\tag{1}
\end{align}

By Cayley–Hamilton, \(f(\vec{A})=\vec{0}\), hence
\begin{align}
\vec{A}^2-\vec{A}=\vec{0}
\;\Longrightarrow\;
\vec{A}^2=\vec{A}.
\tag{2}
\end{align}

Multiplying \((2)\) on the right by \(\vec{A}\) yields
\begin{align}
\vec{A}^3=\vec{A}^2\vec{A}=\vec{A}\vec{A}=\vec{A}^2.
\tag{3}
\end{align}
Using \((2)\) in \((3)\),
\begin{align}
\vec{A}^3=\vec{A}.
\tag{4}
\end{align}
\end{frame}
